cmake_minimum_required(VERSION 3.22)
project(OAPInstalledConsumer LANGUAGES CXX)
find_package(OpenAutoPath CONFIG REQUIRED)
add_executable(oap_installed_consumer main.cpp)
target_link_libraries(oap_installed_consumer PRIVATE
    OAP::Analysis OAP::PocketToolpath OAP::FinishingToolpath OAP::ToolLibrary)
if(MSVC)
    target_compile_options(oap_installed_consumer PRIVATE /utf-8)
endif()

# Each public header must compile as the first and only include in its own TU.
# Only installed paths and imported targets are available in this project.
if(OAP_SDK_INCLUDE_DIR)
    file(GLOB_RECURSE sdk_headers RELATIVE "${OAP_SDK_INCLUDE_DIR}"
        "${OAP_SDK_INCLUDE_DIR}/*.h")
    list(FILTER sdk_headers EXCLUDE REGEX "^thirdparty/")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/sdk-headers")
    set(header_sources)
    foreach(header IN LISTS sdk_headers)
        string(SHA256 header_key "${header}")
        set(source "${CMAKE_CURRENT_BINARY_DIR}/sdk-headers/${header_key}.cpp")
        file(WRITE "${source}" "#include <${header}>\n")
        list(APPEND header_sources "${source}")
    endforeach()
    add_library(oap_installed_headers OBJECT ${header_sources})
    get_property(sdk_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
    list(FILTER sdk_targets INCLUDE REGEX "^OAP::")
    target_link_libraries(oap_installed_headers PRIVATE ${sdk_targets})
    set_target_properties(oap_installed_headers PROPERTIES UNITY_BUILD OFF)
    if(MSVC)
        target_compile_options(oap_installed_headers PRIVATE /utf-8 /MP2 /FS)
    endif()
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND oap_installed_consumer)
